def traducir_senal(texto_original):
    texto_peru = texto_original
    # Reemplazo directo ignorando mayúsculas/minúsculas de forma segura
    for casa_br, casa_pe in DICCIONARIO_CASAS.items():
        patron = re.compile(re.escape(casa_br), re.IGNORECASE)
        texto_peru = patron.sub(casa_pe, texto_peru)
    for termino_br, termino_pe in DICCIONARIO_MERCADOS.items():
        texto_peru = texto_peru.replace(termino_br, termino_pe)
    return texto_peru

def extraer_equipos(texto):
    lineas = texto.split('\n')
    for linea in lineas:
        if ("–" in linea or "-" in linea) and "Fecha" not in linea and "Casas" not in linea and ":" not in linea:
            return linea.strip().replace("➡ ", "")
    return None

async def manejar_mensaje(update: Update, context: ContextTypes.DEFAULT_TYPE):
    texto_recibido = update.message.text
    if not texto_recibido:
        return

    mensaje_traducido = traducir_senal(texto_recibido)
    partido = extraer_equipos(texto_recibido)
    botones = []
    
    if partido:
        query_busqueda = urllib.parse.quote_plus(partido)
        # Enlaces directos a Google para facilitar la búsqueda manual del partido

        url_betsson = f"https://google.com{query_busqueda}+betsson+peru"

            [InlineKeyboardButton("🔍 Buscar en Betano Perú", url=url_betano)],
            [InlineKeyboardButton("🔍 Buscar en Betsson Perú", url=url_betsson)]
        ]
    
    reply_markup = InlineKeyboardMarkup(botones) if botones else None
    
    await update.message.reply_text(
        text=f"🇵🇪 **SEÑAL ADAPTADA PARA PERÚ** 🇵🇪\n\n{mensaje_traducido}",
        reply_markup=reply_markup,
        parse_mode="Markdown"
    )
